module main

data Bad = Broken | Other

fun probe(path: internal Text) : Unit uses {files ↦ Files[write]} fails Bad =
  let _: Unit =
    handle
      let %t0: Unit = files.write("log.txt", "a") in
      %t0
    with error ⇒
      fail Other
  in
  handle
    let %t1: Unit = files.write(path, "x") in
    %t1
  with error%2 ⇒
    fail Broken
